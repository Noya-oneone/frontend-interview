// 模拟 Vue 的响应式系统
function defineReactive(obj, key, val) {
    // 递归处理对象的嵌套属性
    if (typeof val === 'object') {
        observe(val);
    }

    Object.defineProperty(obj, key, {
        enumerable: true,
        configurable: true,
        get() {
            console.log(`Getting property ${key}: ${val}`);
            return val;
        },
        set(newVal) {
            if (newVal === val) {
                return;
            }
            console.log(`Setting property ${key} to: ${newVal}`);
            // 若新值是对象，也进行响应式处理
            if (typeof newVal === 'object') {
                observe(newVal);
            }
            val = newVal;
            // 这里可以添加通知更新视图的逻辑
        }
    });
}

function observe(obj) {
    if (typeof obj!== 'object' || obj === null) {
        return;
    }
    Object.keys(obj).forEach(key => {
        defineReactive(obj, key, obj[key]);
    });
}

// 使用示例
const data = {
    message: 'Hello, Vue 2!',
    nested: {
        value: 1
    }
};

observe(data);

// 获取属性
console.log(data.message);

// 设置属性
data.message = 'New message';

// 处理嵌套属性
console.log(data.nested.value);
data.nested.value = 2;
