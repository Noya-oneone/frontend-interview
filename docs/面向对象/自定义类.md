<!-- 标题
请实现一个 MyEvent，拥有四个方法on,off,once和trigger

题目描述
class MyEvent {​
on(eventName, cb, ...args) {} // 绑定​
off(eventName, cb) {} // 解绑​
once(eventName, cb, ...args) {} // 绑定一次​
trigger(eventName, ...args) {} // 触发事件​
};​ -->

```javascript
class MyEvent {
    constructor() {
        this.events = {};
    }
    // 绑定事件
  on(eventName, cb, ...args) {
    if (!this.events[eventName]) {
      this.events[eventName] = [];
    }
    this.events[eventName].push({ cb, args });
  }

  // 解绑事件
  off(eventName, cb) {
    if (this.events[eventName]) {
      this.events[eventName] = this.events[eventName].filter(event => event.cb !== cb);
    }
  }

  // 绑定一次性事件
  once(eventName, cb, ...args) {
    const onceWrapper = (...onceArgs) => {
      cb(...onceArgs, ...args);
      this.off(eventName, onceWrapper);
    };
    this.on(eventName, onceWrapper, ...args);
  }

  // 触发事件
  trigger(eventName, ...args) {
    if (this.events[eventName]) {
      this.events[eventName].forEach(event => {
        event.cb(...args, ...event.args);
      });
    }
  }   
}

const eventBus = new MyEvent();

eventBus.on('click', (x, y) => {
  console.log(`clicked at ${x}, ${y}`);
});

eventBus.trigger('click', 10, 20); // output: clicked at 10, 20

eventBus.off('click', (x, y) => {
  console.log(`clicked at ${x}, ${y}`);
});

eventBus.trigger('click', 10, 20); // output: no output

eventBus.once('click', (x, y) => {
  console.log(`clicked at ${x}, ${y}`);
});

eventBus.trigger('click', 10, 20); // output: clicked at 10, 20
eventBus.trigger('click', 10, 20); // no output again
```
