# 把二叉搜索树转换为累加树 （BST to Greater Sum Tree）

## 题目描述

* 给定一棵二叉搜索树，请将其转换为累加树。
* 累加树（Greater Sum Tree）是一种树，其中的`每个节点的值`都等于其`左子树中的节点值之和`，`右子树中的节点值之和`。



给出二叉 搜索 树的根节点，该树的节点值各不相同，请你将其转换为累加树（Greater Sum Tree），使每个节点 node 的新值等于原树中大于或等于 node.val 的值之和。

`关键在于反向中序遍历（右 -> 根 -> 左）`

提醒一下，二叉搜索树满足下列约束条件：

节点的左子树仅包含键 小于 节点键的节点。
节点的右子树仅包含键 大于 节点键的节点。
左右子树也必须是二叉搜索树。

## 示例
```js
[4,1,6,0,2,5,7,null,null,null,3,null,null,null,8]
```

转换为累加树后为：

```js
[30,36,21,36,35,26,33,null,null,null,33,null,8]
```

## 代码实现

* 我们可以先遍历二叉搜索树，将每个节点的值更新为其左子树中的节点值之和。然后，我们再遍历二叉搜索树，将每个节点的值更新为其左子树中的节点值之和加上右子树中的节点值之和。

```javascript
function convertBST(root) {
    let total = 0;

    function reverseInOrder(node) {
        if (node === null) return;

        // 先递归遍历右子树
        reverseInOrder(node.right);

        // 更新累积和，并修改当前节点的值
        total += node.val;
        node.val = total;

        // 递归遍历左子树
        reverseInOrder(node.left);
    }

    reverseInOrder(root);
    return root;

}
```
```javascript
// 二叉搜索树的数组表示法
class TreeNode {
    constructor(val = 0, left = null, right = null) {
        this.val = val;
        this.left = left;
        this.right = right;
    }
}

function arrayToBST(arr) {
    if (arr.length === 0) return null;

    const root = new TreeNode(arr[0]);
    const queue = [root];
    let index = 1;

    while (index < arr.length) {
        const node = queue.shift();
        if (index < arr.length && arr[index] !== null) {
            node.left = new TreeNode(arr[index]);
            queue.push(node.left);
        }
        index++;
        if (index < arr.length && arr[index] !== null) {
            node.right = new TreeNode(arr[index]);
            queue.push(node.right);
        }
        index++;
    }

    return root;
}
```

```javascript
// 二叉树在转化成数组
function bstToArray(root) {
    if (root === null) return [];

    const result = [];
    const queue = [root];

    while (queue.length > 0) {
        const node = queue.shift();
        if (node !== null) {
            result.push(node.val);
            queue.push(node.left);
            queue.push(node.right);
        } else {
            result.push(null);
        }
    }

    // Trim trailing nulls for a cleaner output
    while (result[result.length - 1] === null) {
        result.pop();
    }

    return result;
}

// Example usage
const newArr = bstToArray(root);
console.log(newArr);
```
