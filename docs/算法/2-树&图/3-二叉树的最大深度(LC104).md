# 二叉树的最大深度（LeetCode 104）

* 算法题：返回二叉树的`最大深度`（根节点到最远叶子节点的节点数）。
* 前端高频：求 DOM 嵌套深度、JSON 嵌套层级，是递归入门必考题。

## 递归（DFS）- 时间复杂度：O(n)，空间复杂度：O(树高)

```js
function maxDepth(root) {
  if (!root) return 0;
  return 1 + Math.max(maxDepth(root.left), maxDepth(root.right));
}

const tree = {
  val: 3,
  left: { val: 9, left: null, right: null },
  right: {
    val: 20,
    left: { val: 15, left: null, right: null },
    right: { val: 7, left: null, right: null },
  },
};
console.log(maxDepth(tree)); // 3
```

## 迭代（BFS 数层数）- 时间复杂度：O(n)

```js
function maxDepth2(root) {
  if (!root) return 0;
  let depth = 0;
  let queue = [root];
  while (queue.length) {
    const next = [];
    for (const node of queue) {
      if (node.left) next.push(node.left);
      if (node.right) next.push(node.right);
    }
    depth++; // 每处理完一层深度 +1
    queue = next;
  }
  return depth;
}

const tree2 = {
  val: 3,
  left: { val: 9, left: null, right: null },
  right: {
    val: 20,
    left: { val: 15, left: null, right: null },
    right: { val: 7, left: null, right: null },
  },
};
console.log(maxDepth2(tree2)); // 3
```

## 前端变体：求 JSON / DOM 最大嵌套深度

```js
function jsonDepth(obj) {
  if (obj === null || typeof obj !== 'object') return 0;
  const children = Object.values(obj);
  if (!children.length) return 1;
  return 1 + Math.max(...children.map(jsonDepth));
}

console.log(jsonDepth({ a: { b: { c: 1 } }, d: 2 })); // 3
```
