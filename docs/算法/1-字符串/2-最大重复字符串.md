# 求字符串 abcdabcdee 的最大重复字符串

```javascript

function findMaxRepeatingSubstring(str) {
    let maxSubstring = '';
    
    for (let i = 0; i < str.length; i++) {
        for (let j = i + 1; j <= str.length; j++) {
            let substring = str.slice(i, j);
            let count = (str.match(new RegExp(substring, 'g')) || []).length;

            if (count > 1 && substring.length > maxSubstring.length) {
                maxSubstring = substring;
            }
        }
    }

    return maxSubstring;
}

let result = findMaxRepeatingSubstring('abcdabcdee');
console.log(result); // 输出 "abc"
```

```javascript
function longestDupSubstring(s) {
    const n = s.length;
    let left = 1, right = n;
    let result = "";

    while (left < right) {
        const mid = Math.floor((left + right) / 2);
        const dup = search(s, mid, n);

        if (dup) {
            left = mid + 1;
            result = dup;
        } else {
            right = mid;
        }
    }

    return result;
}

function search(s, length, n) {
    const mod = 2 ** 63 - 1;  // Large prime number for mod
    const base = 256;
    let hash = 0;
    const baseL = base ** (length - 1) % mod;
    const seen = new Set();

    // Initial hash calculation for the first window
    for (let i = 0; i < length; i++) {
        hash = (hash * base + s.charCodeAt(i)) % mod;
    }

    seen.add(hash);

    // Sliding window to calculate the hash for subsequent substrings
    for (let i = 1; i <= n - length; i++) {
        hash = (hash * base - s.charCodeAt(i - 1) * baseL % mod + mod) % mod;
        hash = (hash + s.charCodeAt(i + length - 1)) % mod;

        if (seen.has(hash)) {
            return s.slice(i, i + length);
        }
        seen.add(hash);
    }

    return null;
}

let result = longestDupSubstring('abcdabcdee');
console.log(result); // 输出 "abc"
```
