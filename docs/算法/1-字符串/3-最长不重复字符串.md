# 最长不重复字符串
<!-- 滑动窗口技术 -->

```javascript
function lengthOfLongestSubstring(s) {
    let map = new Map();
    let start = 0;
    let maxLength = 0;

    for (let end = 0; end < s.length; end++) {
        let char = s[end];

        if (map.has(char)) {
            // 更新起始位置到重复字符之后
            start = Math.max(map.get(char) + 1, start);
        }

        // 计算当前不重复子字符串的长度
        maxLength = Math.max(maxLength, end - start + 1);

        // 存储当前字符的位置
        map.set(char, end);
    }

    return maxLength;
}

// 示例测试
let result = lengthOfLongestSubstring('abcdabcdee');
console.log(result); // 输出 5, 对应子字符串 "abcde"
```
