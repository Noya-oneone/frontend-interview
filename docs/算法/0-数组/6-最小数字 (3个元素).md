```javascript
function minNumber(arr) {
  // 将数字转换为字符串，方便比较
  // array.sort((a, b) => a - b); a-b 是正序，b-a 是倒序
  // 比较逻辑：a + b - (b + a) 是一种自定义的比较函数，用于确定两个字符串 a 和 b 的排序顺序。
  // 比较函数应该基于拼接后的字符串字典序来决定元素的顺序，而不是元素本身的自然顺序。

  // '3034' 和 '3430'
  // '3034' < '3430'，所以 '30' 应该排在 '34' 前面。
  // const sortedArr = arr.map(String).sort((a, b) => a + b - (b + a));
  const sortedArr = arr.map(String).sort((a, b) => (a + b).localeCompare(b + a));
  // localeCompare：用于比较两个字符串的字典序，确定它们的相对顺序。
  // 返回值：-1、0 或 1，分别表示小于、等于或大于。

  // 选择前 3 个元素，如果数组长度小于 3，则选择所有元素
  const selected = sortedArr.slice(0, Math.min(3, sortedArr.length));

  // 组合成最小数字
  return selected.join("");
}

// 示例输入
const input1 = [21, 30, 62, 5, 31];
const input2 = [5, 21];

// 测试
console.log(minNumber(input1)); // 输出: 21305
console.log(minNumber(input2)); // 输出: 215
```
