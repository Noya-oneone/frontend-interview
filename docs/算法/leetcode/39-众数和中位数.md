# 39-众数和中位数

```javascript
// # 众数和中位数

// 众数是指一组数据中出现次数量多的那个数，众数可以是多个
// 中位数是指把一组数据从小到大排列，最中间的那个数，如果这组数据的个数是
// 奇数，那最中间那个就是中位数
// 如果这组数据的个数为偶数，那就把中间的两个数之和除以 2，所得的结果就是
// 中位数
// 查找整型数组中元素的众数并组成一个新的数组，求新数组的中位数
// 10 11 21 19 21 17 21 16 21 18 16
// 输出：21
// 2 1 5 4 3 3 9 2 7 4 6 2 15 4 2 4
// 输出：3

function findModeAndMedian(arr) {
  const frequency = {}; // 用于记录每个数字出现的频率

  // 计算众数
  arr.forEach((num) => {
    frequency[num] = (frequency[num] || 0) + 1; // 统计出现次数
  });

  // 找到众数
  const maxFreq = Math.max(...Object.values(frequency)); // 最大频率
  const modes = Object.keys(frequency).filter(
    (num) => frequency[num] === maxFreq
  ); // 找出所有众数

  // 计算中位数
  const sortedArr = arr.slice().sort((a, b) => a - b); // 排序数组
  const len = sortedArr.length; // 数组长度
  let median;

  if (len % 2 === 1) {
    // 如果是奇数，取中间的数
    median = sortedArr[Math.floor(len / 2)];
  } else {
    // 如果是偶数，取中间两个数的平均
    median = (sortedArr[len / 2 - 1] + sortedArr[len / 2]) / 2;
  }

  return { modes, median }; // 返回众数和中位数
}

// 示例输入
const input1 = [10, 11, 21, 19, 21, 17, 21, 16, 21, 18, 16];
const result1 = findModeAndMedian(input1);
console.log(result1.modes[0]); // 输出：21

const input2 = [2, 1, 5, 4, 3, 3, 9, 2, 7, 4, 6, 2, 15, 4, 2, 4];
const result2 = findModeAndMedian(input2);
console.log(result2.modes[0]); // 输出：3

```
