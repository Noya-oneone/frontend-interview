# 17-求数组中最大 N 个与最小 N 个数的和，需要对数组进行去重

```javascript
// 说明:数组中数字范围[0, 1000]，最大 N 个数与最小 N 个数不能有重叠，如有重
// 叠，输入非法返回-1
// 第一行输入 M， M 标识数组大小；第二行输入 M 个数，标识数组内容；
// 第三行输入 N，N 表达需要计算的最大、最小 N 个数
// 输出最大 N 个数与最小 N 个数的和
// 5
// 95 88 83 64 100
// 2
// 输出：342

function sumMaxMin(M, arr, N) {
  // 检查输入合法性
  if (N > M || N < 1) return -1; // N 超过数组大小或小于 1

  // 获取最大 N 个数
  const maxNumbers = [...arr].sort((a, b) => b - a).slice(0, N);
  // 获取最小 N 个数
  const minNumbers = [...arr].sort((a, b) => a - b).slice(0, N);

  // 检查最大和最小 N 个数是否有重叠
  const hasOverlap = maxNumbers.some((num) => minNumbers.includes(num));
  if (hasOverlap) return -1; // 有重叠，返回 -1

  // 计算最大和最小 N 个数的和
  const sum =
    maxNumbers.reduce((a, b) => a + b, 0) +
    minNumbers.reduce((a, b) => a + b, 0);

  return sum; // 返回和
}

// 示例输入
const M = 5;
const arr = [95, 88, 83, 64, 100];
const N = 2;
console.log(sumMaxMin(M, arr, N)); // 输出: 342

```
