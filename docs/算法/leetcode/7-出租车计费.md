# 7-出租车计费

```javascript
// 任何数字位置遇到数字 4 就直接跳过，其余功能都正常。
// 输入描述:只有一行，数字 N，表示里程表的读数。
// 输出描述:一个数字，表示实际产生的费用。以回车结束。
// 输入：17
// 输出：15
// 说明：17 表示计费表的表面读数。15 表示实际产生的费用其实只有 15 块钱。
// 5 # 4
// 17 # 15
// 100 # 81

function calculateActualCost(N) {
  let actualCost = 0; // 初始化实际费用计数器
  let count = 0; // 初始化有效数字计数器（可选，不使用）

  // 从 1 遍历到 N
  for (let i = 1; i <= N; i++) {
    // 检查当前数字是否包含 '4'
    if (!String(i).includes("4")) {
      actualCost++; // 如果不包含 '4'，实际费用加一
    }
  }

  return actualCost; // 返回实际产生的费用
}

// 示例输入
const input = 17; // 设置里程表的读数为 17
const output = calculateActualCost(input); // 计算实际费用
console.log(output); // 输出结果: 15

```
