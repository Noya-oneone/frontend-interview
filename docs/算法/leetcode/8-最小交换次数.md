# 8-最小交换次数

```javascript
// 整数数组 nums，整数 k。输出将数组中小于 k 的整数组合到一起的最小交换次数
// 组合在一起是指满足条件的数字相邻，不要求相邻后在数组中的位置
// 1 3 1 4 0 k=2
// 1 解析：交换第一个 1 和 4
// 0 0 0 1 0 k=2
// 0 解析：已经满足条件
// 2 3 2 k=1
// 0 解析：不存在小于 k 的数，无需交换

function minSwaps(nums, k) {
  // 统计数组中小于 k 的元素数量
  const lessThanK = nums.filter((num) => num < k).length;

  // 如果没有小于 k 的元素，直接返回 0
  if (lessThanK === 0) return 0;

  let minSwaps = Infinity; // 初始化最小交换次数为无穷大
  let currentCount = 0; // 当前窗口中小于 k 的元素数量

  // 初始化第一个窗口
  for (let i = 0; i < lessThanK; i++) {
    if (nums[i] < k) currentCount++; // 统计第一个窗口内的小于 k 的元素数量
  }

  // 计算初始窗口的交换次数
  minSwaps = Math.min(minSwaps, lessThanK - currentCount);

  // 滑动窗口
  for (let i = lessThanK; i < nums.length; i++) {
    // 移动窗口，添加新元素
    if (nums[i] < k) currentCount++; // 检查新元素是否小于 k
    // 移除旧元素
    if (nums[i - lessThanK] < k) currentCount--; // 检查移出的元素

    // 更新最小交换次数
    minSwaps = Math.min(minSwaps, lessThanK - currentCount);
  }

  return minSwaps; // 返回最小交换次数
}

// 示例输入
const nums1 = [1, 3, 1, 4, 0];
const k1 = 2;
console.log(minSwaps(nums1, k1)); // 输出：1

const nums2 = [2, 3, 2];
const k2 = 1;
console.log(minSwaps(nums2, k2)); // 输出：0

```
