# 31-组成最大数

```javascript
// # 组成最大数

// 将卡片连起来可以组成多种数字，计算组成的最大数字。
// 22,221
// 输出：22221
// 4589,101,41425,9999
// 输出：9999458941425101

function largestNumber(cards) {
  // 自定义比较函数
  const compare = (a, b) => {
    return (b + a).localeCompare(a + b); // 比较两个拼接后的字符串
  };

  // 排序卡片
  const sortedCards = cards.sort(compare);

  // 处理排序后的结果，若首位为 0，返回 "0"
  if (sortedCards[0] === "0") {
    return "0";
  }

  // 连接所有卡片，形成最大数字
  return sortedCards.join("");
}

// 示例输入
const input1 = "22,221";
const input2 = "4589,101,41425,9999";

console.log(largestNumber(input1.split(","))); // 输出: 22221
console.log(largestNumber(input2.split(","))); // 输出: 9999458941425101

```
