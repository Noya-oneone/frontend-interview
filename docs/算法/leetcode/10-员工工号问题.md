# 10-员工工号问题

```javascript
// 工号系统由小写英文字母（a-z）和数字（0-9）两部分构成。新工号由一段英文
// 字母开头，之后跟随一段数字，比如"aaahw0001",“a12345”,“abcd1”,“a00”
// 注意新工号不能全为字母或者数字,允许数字部分有前导 0 或者全为 0。
// 但是过长的工号会增加同事们的记忆成本，现在给出新工号至少需要分配的人数
// X 和新工号中字母的长度 Y
// 求新工号中数字的最短长度 Z。
// 一行两个非负整数 X Y，数字用单个空格分隔
// 输出新工号中数字的最短长度 Z
// 260 1 # 1
// 26 1 # 1
// 2600 1 # 2

function minDigitLength(X, Y) {
  // 计算字母部分的组合数
  const letterCombinations = Math.pow(26, Y); // 小写字母组合数

  // 计算总工号的组合数
  const totalCombinations = letterCombinations * Math.pow(10, Z); // 总组合数

  // 初始化数字部分长度
  let Z = 0;

  // 查找最小数字长度 Z
  while (totalCombinations < X) {
    Z++; // 增加数字长度
    totalCombinations = letterCombinations * Math.pow(10, Z); // 更新总组合数
  }

  return Z; // 返回最小数字长度
}

// 示例输入
console.log(minDigitLength(260, 1)); // 输出: 1
console.log(minDigitLength(26, 1)); // 输出: 1
console.log(minDigitLength(2600, 1)); // 输出: 2

```
