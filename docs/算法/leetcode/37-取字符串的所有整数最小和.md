# 37-取字符串的所有整数最小和

```javascript
// # 取字符串的所有整数最小和
// 输入字符串 s，输出 s 中包含所有整数的最小和
// 说明：字符串 s，只包含 a-z A-Z ± ；
// 合法的整数包括：1） 正整数 一个或者多个 0-9 组成，如 0 2 3 002 102
// 2）负整数 负号 - 开头，数字部分由一个或者多个 0-9 组成
// bb1234aa
// 输出：10
// bb12-34aa
// 输出：-31

function minSumOfIntegers(s) {
  const regex = /[+-]?\d+/g; // 正则表达式匹配整数
  const matches = s.match(regex) || []; // 找到所有匹配的整数

  // 计算所有整数的和
  const sum = matches.reduce((acc, num) => acc + parseInt(num, 10), 0);

  return sum; // 返回最小和
}

// 示例输入
const input1 = "bb1234aa";
console.log(minSumOfIntegers(input1)); // 输出：10

const input2 = "bb12-34aa";
console.log(minSumOfIntegers(input2)); // 输出：-31

```
