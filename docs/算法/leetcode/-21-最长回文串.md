# -21-最长回文串

```javascript
// 最长回文串 https://leetcode.cn/problems/longest-palindrome/
// 回文子字符串是指一个字符串中的某个子串，该子串从前往后读和从后往前读是一样的
// "aba" 是回文子字符串，因为反向读取也是 "aba"。
// "racecar" 是回文子字符串，因为反向读取也是 "racecar"。
// "abc" 不是回文子字符串，因为反向读取是 "cba"。

// 给定一个包含大写字母和小写字母的字符串 s ，返回 通过这些字母构造成的 最长的
// 回文串
//  的长度。

// 在构造过程中，请注意 区分大小写 。比如 "Aa" 不能当做一个回文字符串。

// 示例 1:

// 输入:s = "abccccdd"
// 输出:7
// 解释:
// 我们可以构造的最长的回文串是"dccaccd", 它的长度是 7。
// 示例 2:

// 输入:s = "a"
// 输出:1
// 解释：可以构造的最长回文串是"a"，它的长度是 1。

// 提示:

// 1 <= s.length <= 2000
// s 只由小写 和/或 大写英文字母组成

// https://leetcode.cn/problems/longest-palindrome
//  最长回文串
/**
 * @param {string} s
 * @return {number}
 */
var longestPalindrome = function (s) {
  const charCount = {};
  let length = 0;
  let hasOdd = false;

  // 统计每个字符的出现次数
  for (const char of s) {
    charCount[char] = (charCount[char] || 0) + 1;
  }

  // 遍历字符计数，计算回文长度
  for (const count of Object.values(charCount)) {
    length += Math.floor(count / 2) * 2; // 加入偶数部分
    if (count % 2 === 1) {
      hasOdd = true; // 检查是否有奇数个字符
    }
  }

  // 如果有奇数个字符，可以在中间放一个
  return hasOdd ? length + 1 : length;
};

```
