# -8-*HJ23.删除字符串中出现次数最少的字符

```javascript
// 实现删除字符串中出现次数最少的字符，
// 若出现次数最少的字符有多个，则把出现次数最少的字符都删除。
// 出删除这些单词后的字符串，字符串中其它字符保持原来的顺序。

// 输入的字符串长度满足 1 ≤ n ≤ 20，保证输入的字符串中仅出现小写字母

// 字符串只包含小写英文字母, 不考虑非法输入，输入的字符串长度小于等于20个字节
function removeLeastFrequentChar(s) {
  if (s.length < 1 || s.length > 20) {
    throw new Error("输入字符串长度必须在 1 到 20 之间");
  }
  if (!/^[a-z]+$/.test(s)) {
    throw new Error("输入字符串必须只包含小写字母");
  }
  const count = {};

  // 统计字符出现次数
  for (const char of s) {
    count[char] = (count[char] || 0) + 1;
  }

  // 找到最少出现的字符
  const minFreq = Math.min(...Object.values(count));
  const leastFrequentChars = new Set(
    Object.keys(count).filter((char) => count[char] === minFreq)
  );

  // 生成新字符串，去除最少出现的字符
  const result = Array.from(s)
    .filter((char) => !leastFrequentChars.has(char))
    .join("");

  return result;
}

// 示例输入
const inputStr = "abcabcdd";
const outputStr = removeLeastFrequentChar(inputStr);
console.log(outputStr); // 输出：abcabc

```
