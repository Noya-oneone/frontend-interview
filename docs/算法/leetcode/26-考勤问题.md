# 26-考勤问题

```javascript
// 能获得出勤奖的条件如下：缺勤不超过一次； 没有连续的迟到/早退； 任意连
// 续 7 次考勤，缺勤/迟到/早退不超过 3 次
// 2
// present
// present absent present present leaveearly present absent
// 输出描述:根据考勤数据字符串，如果能得到考勤奖，输出"true"；否则输出
// "false"，对于输入示例的结果应为：
// true false

function canGetAttendanceAward(n, records) {
  const attendance = records.split(" "); // 分割考勤记录
  let absentCount = 0; // 缺勤计数
  let lateCount = 0; // 迟到计数
  let earlyLeaveCount = 0; // 早退计数
  let consecutiveLateOrEarly = false; // 标记是否有连续迟到/早退
  let recentAttendance = []; // 用于检查最近 7 次考勤

  for (let i = 0; i < attendance.length; i++) {
    const status = attendance[i];

    // 统计缺勤
    if (status === "absent") {
      absentCount++;
      recentAttendance.push("absent");
    } else if (status === "late") {
      lateCount++;
      recentAttendance.push("late");
      if (
        recentAttendance.length > 1 &&
        recentAttendance[recentAttendance.length - 2] === "late"
      ) {
        consecutiveLateOrEarly = true; // 连续迟到
      }
    } else if (status === "leaveearly") {
      earlyLeaveCount++;
      recentAttendance.push("leaveearly");
      if (
        recentAttendance.length > 1 &&
        recentAttendance[recentAttendance.length - 2] === "leaveearly"
      ) {
        consecutiveLateOrEarly = true; // 连续早退
      }
    } else {
      recentAttendance.push("present");
    }

    // 检查最近 7 次考勤
    if (recentAttendance.length > 7) {
      const removed = recentAttendance.shift(); // 移除最早的记录
      if (removed === "absent") absentCount--;
      if (removed === "late") lateCount--;
      if (removed === "leaveearly") earlyLeaveCount--;
    }

    // 检查考勤奖的条件
    if (
      absentCount > 1 ||
      consecutiveLateOrEarly ||
      (recentAttendance.length === 7 &&
        absentCount + lateCount + earlyLeaveCount > 3)
    ) {
      return "false"; // 不符合条件
    }
  }

  return "true"; // 符合条件
}

// 示例输入
const n1 = 2;
const records1 =
  "present present absent present present leaveearly present absent";
console.log(canGetAttendanceAward(n1, records1)); // 输出: true

const n2 = 2;
const records2 = "present present late present leaveearly present absent late";
console.log(canGetAttendanceAward(n2, records2)); // 输出: false

```
