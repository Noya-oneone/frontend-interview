# 3-找终点(最小步骤数)

```javascript
// 一个正整数数组设为 nums,最大为 100 个成员。求从第一个成员开始正好走到数组最后一个成员所使用的最小步骤数
// 要求：1.第一步必须从第一元素起，且 1<=第一步步长<len/2 (len 为数组长
// 度)
// 2.从第二步开始只能以所在成员的数字走相应的步数，不能多不能少，如果目标
// 不可达返回-1，只输出最小的步骤数量
// 3.只能向数组的尾部走不能向回走
// 输入描述：有正整数数组，空格分割
// 输出描述 ：最小步数，不存在输出-1
// 7 5 9 4 2 6 8 3 5 4 3 9
// 2
// 1 2 3 7 1 5 9 3 2 1
// -1

function minSteps(nums) {
  const len = nums.length;

  // 初始化步数数组，-1表示不可达
  const steps = new Array(len).fill(-1);
  steps[0] = 0; // 起始位置步数为0

  // 第一阶段：计算第一步的可行步长
  for (let firstStep = 1; firstStep < Math.min(len / 2, len); firstStep++) {
    if (firstStep < len) {
      steps[firstStep] = 1; // 第一轮的步数
    }
  }

  // BFS-like遍历
  for (let i = 0; i < len; i++) {
    if (steps[i] === -1) continue; // 如果当前不可达，跳过

    // 从当前位置开始，使用当前数字的步数前进
    const jump = nums[i];

    for (let j = i + jump; j < len; j += jump) {
      if (steps[j] === -1) {
        steps[j] = steps[i] + 1; // 更新步数
      }
    }
  }

  return steps[len - 1] !== -1 ? steps[len - 1] : -1; // 返回最后一个元素的步数或-1
}

// 示例输入
const input1 = [7, 5, 9, 4, 2, 6, 8, 3, 5, 4, 3, 9];
const input2 = [1, 2, 3, 7, 1, 5, 9, 3, 2, 1];

console.log(minSteps(input1)); // 输出最小步数
console.log(minSteps(input2)); // 输出 -1

```
