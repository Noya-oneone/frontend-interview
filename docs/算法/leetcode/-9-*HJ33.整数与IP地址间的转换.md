# -9-*HJ33.整数与IP地址间的转换

```javascript
// # *HJ33.整数与IP地址间的转换

// 描述
// 原理：ip地址的每段可以看成是一个0-255的整数，把每段拆分成一个二进制形式组合起来，然后把这个二进制数转变成
// 一个长整数。
// 举例：一个ip地址为10.0.3.193
// 每段数字             相对应的二进制数
// 10                   00001010
// 0                    00000000
// 3                    00000011
// 193                  11000001

// 组合起来即为：00001010 00000000 00000011 11000001,转换为10进制数就是：167773121，即该IP地址转换后的数字就是它了。

// 数据范围：保证输入的是合法的 IP 序列

// 输入描述：
// 输入
// 1 输入IP地址
// 2 输入10进制型的IP地址

// 输出描述：
// 输出
// 1 输出转换成10进制的IP地址
// 2 输出转换后的IP地址

const rl = require("readline").createInterface({ input: process.stdin });
const iter = rl[Symbol.asyncIterator]();

const readline = async () => (await iter.next()).value;

function ipToDecimal(ip) {
  const segments = ip.split(".").map(Number);
  return (
    (segments[0] << 24) |
    (segments[1] << 16) |
    (segments[2] << 8) |
    (segments[3] >>> 0)
  );
}

function decimalToIp(decimal) {
  return [
    (decimal >>> 24) & 255,
    (decimal >>> 16) & 255,
    (decimal >>> 8) & 255,
    decimal & 255,
  ].join(".");
}

(async () => {
  // 输入 IP 地址
  const ipInput = (await readline()).trim();
  const decimalOutput = ipToDecimal(ipInput);
  console.log(decimalOutput); // 输出转换成10进制的IP地址

  // 输入10进制型的IP地址
  const decimalInput = parseInt((await readline()).trim(), 10);
  const ipOutput = decimalToIp(decimalInput);
  console.log(ipOutput); // 输出转换后的IP地址

  rl.close();
})();

```
