# 13-最大消费金额

```javascript
// 从众多心仪商品中购买 3 件，而且想尽可能的花完资金，现在请你设计一个程
// 序，计算小明尽可能花费的最大资金数
// 输入第一行为一维整型数组 m，数组元素记录单个商品的价格
// 输入第二行为购买资金的额度 r
// 输出为满足上述条件的最大花费额度。注意：如果不存在满足上述条件的商品请
// 返回-1
// 23,26,36,27
// 78
// 输出 76
// 23,30,40
// 26
// 输出 -1
function maxSpending(prices, budget) {
  const n = prices.length; // 获取商品数量
  let maxSpend = -1; // 初始化最大花费为 -1，表示无解

  // 三重循环选取三件商品
  for (let i = 0; i < n; i++) {
    for (let j = i + 1; j < n; j++) {
      for (let k = j + 1; k < n; k++) {
        const total = prices[i] + prices[j] + prices[k]; // 计算当前三件商品的总价
        if (total <= budget) {
          maxSpend = Math.max(maxSpend, total); // 更新最大花费
        }
      }
    }
  }

  return maxSpend; // 返回最大花费或 -1
}

// 示例输入
console.log(maxSpending([23, 26, 36, 27], 78)); // 输出: 76
console.log(maxSpending([23, 30, 40], 26)); // 输出: -1

```
