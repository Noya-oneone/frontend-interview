# -10-HJ101.输入整型数组和排序标识

```javascript
// # HJ101.输入整型数组和排序标识

// 输入整型数组和排序标识，对其元素按照升序或降序进行排序

// 数据范围： 1≤n≤1000 1≤n≤1000 ，元素大小满足 0≤val≤100000 0≤val≤100000

//  输入描述：
//  第一行输入元素个数
//   第二行输入待排序的数组，每个数用空格隔开
//  第三行输入一个整数0或1。0代表升序排序，1代表降序排序

//  输出描述
//  输出排好序的数字

// 8
// 1 2 4 9 3 55 64 25
//  0

const rl = require("readline").createInterface({ input: process.stdin });
const iter = rl[Symbol.asyncIterator]();

const readline = async () => (await iter.next()).value;

(async () => {
  // 输入元素个数
  const n = parseInt((await readline()).trim(), 10);

  // 输入待排序的数组
  const array = (await readline()).trim().split(" ").map(Number);

  // 输入排序标识
  const order = parseInt((await readline()).trim(), 10);

  // 排序
  array.sort((a, b) => (order === 0 ? a - b : b - a));

  // 输出结果
  console.log(array.join(" "));

  rl.close();
})();

```
