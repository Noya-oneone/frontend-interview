# 12-整数对最小和

```javascript
// 给定两个整数数组 array1、array2，数组元素按升序排列
// 假设从 array1、array2 中分别取出一个元素可构成一对元素，现在需要取出 k
// 对元素
// 并对取出的所有元素求和，计算和的最小值
// 3 1 1 2
// 3 1 2 3
// 2
// 输出 4

function minSum(array1, array2, k) {
  // 先排序两个数组（确保它们是升序的）
  array1.sort((a, b) => a - b);
  array2.sort((a, b) => a - b);

  let sum = 0; // 初始化和为 0

  // 取出 k 对元素，选择最小的元素对
  for (let i = 0; i < k; i++) {
    sum += array1[i] + array2[i]; // 将对应的元素相加
  }

  return sum; // 返回计算出的最小和
}

// 示例输入
console.log(minSum([1, 1, 2, 3], [1, 2, 3], 2)); // 输出: 4

```
