# 11-身高体重排序

```javascript
// 按照身高由低到高排列，对身高相同的人，按体重由轻到重排列；对于身高体重
// 都相同的人，维持原有的编号顺序关系。
// 请输出排列后的学生编号。
// 第一个序列中的数值代表身高，第二个序列中的数值代表体重。
// 输出描述:排列结果，每个数值都是原始序列中的学生编号，编号从 1 开始
// 4
// 100 100 120 130
// 40 30 60 50
// 输出：2134
// 3
// 90 110 90
// 45 60 45
// 输出：132

function sortStudents(heights, weights) {
  const students = heights.map((height, index) => ({
    index: index + 1, // 学生编号从1开始
    height,
    weight: weights[index],
  }));

  // 排序：首先按身高升序，其次按体重升序
  students.sort((a, b) => {
    if (a.height === b.height) {
      return a.weight - b.weight; // 体重升序
    }
    return a.height - b.height; // 身高升序
  });

  // 提取排序后的学生编号
  return students.map((student) => student.index).join(""); // 输出编号
}

// 示例输入
console.log(sortStudents([100, 100, 120, 130], [40, 30, 60, 50])); // 输出: 2134
console.log(sortStudents([90, 110, 90], [45, 60, 45])); // 输出: 132

```
