# 20-报数游戏

```javascript
// 100 个人围成一圈，每个人有一个编码，编号从 1 开始到 100。他们从 1 开始依
// 次报数，报到为 M 的人自动退出圈圈，
// 然后下一个人接着从 1 开始报数，直到剩余的人数小于 M。请问最后剩余的人在
// 原先的编号为多少？
// 输入一个整数参数 M
// 输出描述:如果输入参数 M 小于等于 1 或者大于等于 100，输出“ERROR!”；否
// 则按照原先的编号从小到大的顺序，以英文逗号分割输出编号字符串
// 3
// 输出：58,91

function lastRemaining(M) {
  // 输入参数验证
  if (M <= 1 || M >= 100) return "ERROR!";

  const people = Array.from({ length: 100 }, (_, i) => i + 1); // 创建 1 到 100 的数组
  let index = 0; // 当前报数的索引

  // 循环直到剩余人数小于 M
  while (people.length >= M) {
    index = (index + M - 1) % people.length; // 计算报到 M 的人索引
    people.splice(index, 1); // 移除报到的人
  }

  return people.join(","); // 返回剩余人的编号字符串
}

// 示例输入
const M = 3;
console.log(lastRemaining(M)); // 输出: 58,91

```
