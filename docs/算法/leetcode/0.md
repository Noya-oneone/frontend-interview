function removeDuplicates(nums) {
  if (nums.length === 0) return 0;
  let i = 0; // i 是一个指针，表示当前唯一元素的位置的索引，初始值为 0。
  for (let j = 1; j < nums.length; j++) {
    // j 是另一个指针，用于遍历数组
    // 如果当前元素与前一个元素不同
    if (nums[i] !== nums[j]) {
      i++; // 移动唯一元素的位置
      nums[i] = nums[j]; // 将当前新元素 nums[j] 放置到 i 的位置，这样 nums[0] 到 nums[i] 都是唯一的元素。
    }
  }
  console.log(nums.slice(0, i + 1)); // 打印去重后的数组
  return i + 1; // 返回去重后的数组长度
}

console.log(removeDuplicates([1, 1, 1, 1, 1, 1, 2, 4, 4, 4444, 5555]));
console.log();
