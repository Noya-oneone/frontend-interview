# 27-字符串的去重和排序

```javascript
// 给定两个字符串，从字符串 2 中找出字符串 1 中的所有字符
// 去重并按照 ASCII 码值从小到大排列
// bach
// bbaaccddfg
// 输出：abc

function findUniqueChars(str1, str2) {
  const uniqueChars = new Set(); // 使用 Set 来存储唯一字符

  // 遍历字符串 2
  for (const char of str2) {
    if (str1.includes(char)) {
      // 如果字符在字符串 1 中
      uniqueChars.add(char); // 添加到 Set 中
    }
  }

  // 将 Set 转换为数组并排序
  const sortedChars = Array.from(uniqueChars).sort();

  // 返回排序后的字符连接成字符串
  return sortedChars.join("");
}

// 示例输入
const str1 = "bach";
const str2 = "bbaaccddfg";
console.log(findUniqueChars(str1, str2)); // 输出: abc

```
