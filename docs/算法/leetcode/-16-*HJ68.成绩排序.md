# -16-*HJ68.成绩排序

```javascript
// # *HJ68.成绩排序
// 描述: 查找和排序

// 题目：输入任意（用户，成绩）序列，可以获得成绩从高到低或从低到高的排列,相同成绩
// 都按先录入排列在前的规则处理。
const readline = require("readline");

// 创建接口实例
const rl = readline.createInterface({
  input: process.stdin,
  output: process.stdout,
});

let data = [];
let currentIndex = 0;

// 读取输入
rl.on("line", (line) => {
  if (currentIndex === 0) {
    currentIndex++;
    return; // 忽略首行
  }

  const input = line.trim();

  // 如果是排序指令
  if (input === "0" || input === "1") {
    const order = parseInt(input);
    const numEntries = data.length;

    // 排序
    data.sort((a, b) => {
      if (a.score === b.score) {
        return 0; // 相同分数时保持原顺序
      }
      return order === 0 ? b.score - a.score : a.score - b.score; // 降序或升序
    });

    // 输出结果
    for (let entry of data) {
      console.log(`${entry.name} ${entry.score}`);
    }

    // 清空数据以处理下一组输入
    data = [];
  } else {
    // 处理学生姓名和成绩输入
    const [name, score] = input.split(" ");
    data.push({ name, score: parseInt(score) });
  }
});

// 处理结束事件
rl.on("close", () => {
  console.log("输入结束");
});

```
