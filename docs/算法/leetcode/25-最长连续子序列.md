# 25-最长连续子序列

```javascript
// 有 n 个正整数组成的一个序列。给定正整数 sum,求长度最长的连续子序列，使
// 得他们的和等于 sum。
// 返回此子序列的长度，如果没有要求的序列，返回-1
// 1,2,3,4,2
// 6
// 输出：3
// 解析：1,2,3 和 4,2 两个序列均能满足要求，所以最长的连续序列为 1,2,3 因此
// 结果为 3
// 1,2,3,4,2
// 20
// 输出：-1

function longestSubarrayWithSum(nums, targetSum) {
  const n = nums.length;
  let maxLength = -1; // 初始化最大长度为 -1
  let sumMap = new Map(); // 用于存储前缀和及其索引
  let currentSum = 0; // 当前前缀和

  // 遍历数组
  for (let i = 0; i < n; i++) {
    currentSum += nums[i]; // 更新当前前缀和

    // 检查当前前缀和是否等于目标和
    if (currentSum === targetSum) {
      maxLength = i + 1; // 更新最大长度
    }

    // 如果前缀和减去目标和存在于 map 中，更新最大长度
    if (sumMap.has(currentSum - targetSum)) {
      const subArrayLength = i - sumMap.get(currentSum - targetSum);
      maxLength = Math.max(maxLength, subArrayLength); // 更新最大长度
    }

    // 记录前缀和及其索引
    if (!sumMap.has(currentSum)) {
      sumMap.set(currentSum, i); // 仅记录第一次出现的前缀和
    }
  }

  return maxLength; // 返回最大长度或 -1
}

// 示例输入
const nums1 = [1, 2, 3, 4, 2];
const targetSum1 = 6;
console.log(longestSubarrayWithSum(nums1, targetSum1)); // 输出: 3

const nums2 = [1, 2, 3, 4, 2];
const targetSum2 = 20;
console.log(longestSubarrayWithSum(nums2, targetSum2)); // 输出: -1

```
