# -18-*括号的最大嵌套深度

```javascript
// # *leetcode 1614.括号的最大嵌套深度

// 给定 有效括号字符串 s，返回 s 的 嵌套深度。嵌套深度是嵌套括号的 最大 数量。

// 示例 1：

// 输入：s = "(1+(2*3)+((8)/4))+1"

// 输出：3

// 解释：数字 8 在嵌套的 3 层括号中。

// 示例 2：

// 输入：s = "(1)+((2))+(((3)))"

// 输出：3

// 解释：数字 3 在嵌套的 3 层括号中。

// 示例 3：

// 输入：s = "()(())((()()))"

// 输出：3

// 提示：

// 1 <= s.length <= 100
// s 由数字 0-9 和字符 '+'、'-'、'*'、'/'、'('、')' 组成
// 题目数据保证括号字符串 s 是 有效的括号字符串

// https:leetcode.cn/problems/maximum-nesting-depth-of-the-parentheses/description/

function maxDepth(s) {
  let depth = 0; // 当前嵌套深度
  let maxDepth = 0; // 最大嵌套深度

  for (let char of s) {
    if (char === "(") {
      depth++; // 遇到左括号，深度增加
      maxDepth = Math.max(maxDepth, depth); // 更新最大深度
    } else if (char === ")") {
      depth--; // 遇到右括号，深度减少
    }
  }

  return maxDepth; // 返回最大嵌套深度
}

```
