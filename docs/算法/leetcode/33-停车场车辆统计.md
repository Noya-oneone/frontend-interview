# 33-停车场车辆统计

```javascript
// # 停车场车辆统计

// 特定大小的停车场，用数组 cars 表示，其中 1 表示有车 0 表示没车。
// 车辆大小不一，小车占一个车位(长度 1)，货车占两个车位(长度 2)，卡车占三
// 个车位(长度 3)。
// 统计停车场最少可以停多少辆车，返回具体的数目。
// 整型字符串数组 cars，其中 1 表示有车 0 表示没车，数组长度<1000
// 整型数字字符串，表示最少停车数
// 1,1,0,0,1,1,1,0,1
// 输出：3

function minCars(cars) {
  let count = 0; // 统计停车数量
  let i = 0; // 遍历停车场

  while (i < cars.length) {
    if (cars[i] === "1") {
      count++; // 有车，停车数量加1
      i++; // 小车占1个车位，移动到下一个车位
    } else if (i < cars.length - 1 && cars[i] === "0" && cars[i + 1] === "1") {
      count++; // 遇到货车，占2个车位
      i += 2; // 移动到下一个可用车位
    } else if (
      i < cars.length - 2 &&
      cars[i] === "0" &&
      cars[i + 1] === "0" &&
      cars[i + 2] === "1"
    ) {
      count++; // 遇到卡车，占3个车位
      i += 3; // 移动到下一个可用车位
    } else {
      i++; // 如果没有车，继续检查下一个车位
    }
  }

  return count; // 返回最少停车数
}

// 示例输入
const input = "1,1,0,0,1,1,1,0,1";
const cars = input.split(",");

console.log(minCars(cars)); // 输出: 3

```
