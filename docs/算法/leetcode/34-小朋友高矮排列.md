# 34-小朋友高矮排列

```javascript
// # 小朋友高矮排列

// 以“高”“矮”“高”“矮”顺序排列，每一个“高”位置的小朋友要比相邻的位
// 置高或者相等；
// 每一个“矮”位置的小朋友要比相邻的位置矮或者相等；
// 要求小朋友们移动的距离和最小，第一个从“高”位开始排，输出最小移动距离
// 即可。
// 4 1 3 5 2
// 输出：4 1 5 2 3

function arrangeStudents(heights) {
  // 将身高排序
  const sortedHeights = heights.slice().sort((a, b) => a - b);
  const n = heights.length;
  const result = new Array(n); // 初始化结果数组

  // 按照“高矮高矮”的顺序填充结果
  for (let i = 0; i < n; i++) {
    if (i % 2 === 0) {
      result[i] = sortedHeights[Math.floor(i / 2)]; // 高的位置
    } else {
      result[i] = sortedHeights[Math.floor(n / 2) + Math.floor(i / 2)]; // 矮的位置
    }
  }

  // 计算移动距离
  const distance = heights.reduce((sum, height, index) => {
    return sum + Math.abs(height - result[index]);
  }, 0);

  return { arranged: result, distance: distance };
}

// 示例输入
const heights = [4, 1, 3, 5, 2];
const { arranged, distance } = arrangeStudents(heights);

console.log(arranged.join(" ")); // 输出排列后的身高
console.log("最小移动距离:", distance); // 输出最小移动距离

```
