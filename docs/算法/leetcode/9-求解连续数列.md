# 9-求解连续数列

```javascript
// 已知连续正整数数列 {K}=K1,K2,K3…Ki 的各个数相加之和为 S ， i=N
// (0<S<100000, 0<N<100000), 求此数列 K
// 输入包含两个参数: 1）连续正整数数列和 S 2）数列里数的个数 N
// 如果有解输出数列 K，如果无解输出-1
// 525 6
// 85 86 87 88 89 90
// 3 5
// -1

function findConsecutiveNumbers(S, N) {
  // 检查输入有效性
  if (S <= 0 || N <= 0 || S >= 100000 || N >= 100000) {
    return -1; // 输入超出范围，返回 -1
  }

  // 计算连续正整数的起始值
  let start = (S - (N * (N - 1)) / 2) / N;

  // 检查起始值是否为正整数
  if (start <= 0 || !Number.isInteger(start)) {
    return -1; // 无解情况，返回 -1
  }

  // 生成连续正整数数列
  const result = [];
  for (let i = 0; i < N; i++) {
    result.push(start + i); // 添加每个连续正整数到结果数组
  }

  return result; // 返回找到的连续正整数数列
}

// 示例输入
console.log(findConsecutiveNumbers(525, 6)); // 输出: [85, 86, 87, 88, 89, 90]
console.log(findConsecutiveNumbers(3, 5)); // 输出: -1

```
