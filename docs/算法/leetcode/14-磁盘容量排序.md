# 14-磁盘容量排序

```javascript
// 磁盘的容量单位常用的有 M，G，T 这三个等级，关系为 1T = 1024G、1G = 1024M
// 如样例所示先输入磁盘的个数，再依次输入磁盘的容量大小，然后按照从小到大
// 的顺序对磁盘容量进行排序并输出。
// 3
// 1G
// 2G
// 1024M
// 输出
// 1G
// 1024M
// 2G
// 3
// 2G4M
// 3M2G
// 1T
// 输出
// 3M2G
// 2G4M
// 1T

function convertToMB(size) {
  // 将磁盘容量转换为 MB
  const value = parseInt(size.slice(0, -1)); // 获取数值部分
  const unit = size.slice(-1); // 获取单位

  switch (unit) {
    case "M":
      return value; // MB直接返回
    case "G":
      return value * 1024; // G转换为 MB
    case "T":
      return value * 1024 * 1024; // T转换为 MB
    default:
      return 0; // 无效单位
  }
}

function formatSize(sizeInMB) {
  // 将 MB 转换回原始格式
  if (sizeInMB >= 1024 * 1024) {
    return sizeInMB / (1024 * 1024) + "T";
  } else if (sizeInMB >= 1024) {
    return sizeInMB / 1024 + "G";
  } else {
    return sizeInMB + "M";
  }
}

function sortDisks(diskSizes) {
  // 转换并排序磁盘容量
  const sortedSizes = diskSizes
    .map((size) => ({ original: size, sizeInMB: convertToMB(size) })) // 生成对象
    .sort((a, b) => a.sizeInMB - b.sizeInMB) // 按照 MB 排序
    .map((obj) => obj.original); // 返回原始大小

  return sortedSizes; // 返回排序后的结果
}

// 示例输入
const diskCount1 = 3;
const diskSizes1 = ["1G", "2G", "1024M"];
console.log(sortDisks(diskSizes1).join("\n")); // 输出结果

const diskCount2 = 3;
const diskSizes2 = ["3M2G", "2G4M", "1T"];
console.log(sortDisks(diskSizes2).join("\n")); // 输出结果

```
