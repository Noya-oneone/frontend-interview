# 23-两数之和绝对值最小

```javascript
// 给定一个随机的整数数组(可能存在正整数和负整数)nums,请你在该数组中找出
// 两个数，其和的绝对值(|nums[x]+nums[y]|)为最小值
// 并返回这两个数(按从小到大返回)以及绝对值。每种输入只会对应一个答案。但
// 是，数组中同一个元素不能使用两遍。
// -1 -3 7 5 11 15
// 输出：-3 5 2
// 说明：因为|nums[0]+nums[2]|=|-3+5|=2 最小，所以返回-3 5 2

function minAbsSumPair(nums) {
  nums.sort((a, b) => a - b); // 排序数组
  let left = 0; // 左指针
  let right = nums.length - 1; // 右指针
  let minAbs = Infinity; // 初始化最小绝对值
  let result = []; // 存储结果

  // 使用双指针查找最小绝对值和
  while (left < right) {
    const sum = nums[left] + nums[right]; // 计算当前和
    const absSum = Math.abs(sum); // 计算绝对值

    // 更新最小绝对值和及其对应的数
    if (absSum < minAbs) {
      minAbs = absSum;
      result = [nums[left], nums[right]];
    }

    // 移动指针
    if (sum < 0) {
      left++; // 如果和为负，左指针右移
    } else {
      right--; // 如果和为正，右指针左移
    }
  }

  return [...result.sort((a, b) => a - b), minAbs]; // 返回结果和最小绝对值
}

// 示例输入
const nums = [-1, -3, 7, 5, 11, 15];
console.log(minAbsSumPair(nums)); // 输出: [-3, 5, 2]

```
