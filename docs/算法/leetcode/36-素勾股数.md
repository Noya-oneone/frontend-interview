# 36-素勾股数

```javascript
// # 素勾股数

// 如果三个正整数 A B C ,A²+B²=C²则为勾股数
// 如果 ABC 之间两两互质，即 A 与 B A 与 C B 与 C 均互质没有公约数，
// 则称其为勾股数元组。
// 请求出给定 n m 范围内所有的勾股数元组
// 1
// 20
// 输出
// 3 4 5
// 5 12 13
// 8 15 17

function gcd(a, b) {
  while (b !== 0) {
    [a, b] = [b, a % b]; // 计算最大公约数
  }
  return a; // 返回最大公约数
}

function isCoprime(a, b) {
  return gcd(a, b) === 1; // 判断两数是否互质
}

function findPythagoreanTriples(n, m) {
  const results = []; // 存储结果

  // 遍历所有可能的三元组 (A, B, C)
  for (let a = 1; a <= m; a++) {
    for (let b = a + 1; b <= m; b++) {
      for (let c = b + 1; c <= m; c++) {
        if (a * a + b * b === c * c) {
          // 判断是否为勾股数
          // 检查三数之间是否两两互质
          if (isCoprime(a, b) && isCoprime(a, c) && isCoprime(b, c)) {
            results.push([a, b, c]); // 添加到结果
          }
        }
      }
    }
  }

  return results; // 返回所有找到的勾股数元组
}

// 示例输入
const n = 1; // 起始值
const m = 20; // 结束值
const triples = findPythagoreanTriples(n, m);

// 输出结果
triples.forEach((triple) => console.log(triple.join(" ")));

```
